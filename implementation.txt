```python
"""
Phase 1 Task 1: Forum Distribution
Count posts per forum and generate distribution summary.
"""

import json
import pandas as pd
from pathlib import Path
from datetime import datetime

PROJECT_ROOT = Path(__file__).resolve().parent.parent
DATA_DIR = PROJECT_ROOT / "data" / "raw"
RESULTS_DIR = PROJECT_ROOT / "results" / "phase1"

# Forum definitions
FORUMS = [
    {"name": "Stack Overflow", "file": "stackoverflow_posts.json", "date_field": "creation_date"},
    {"name": "GitHub Issues", "file": "github_issues.json", "date_field": "created_at"},
    {"name": "Bugzilla", "file": "bugzilla_posts.json", "date_field": "creation_time"},
    {"name": "Microsoft Q&A", "file": "microsoft_qa_posts.json", "date_field": "creationDate"},
    {"name": "Oracle Community", "file": "oracle_community_posts.json", "date_field": "post_date"},
    {"name": "Reddit", "file": "reddit_posts.json", "date_field": "created_utc"},
    {"name": "Hacker News", "file": "hacker_news_posts.json", "date_field": "time"},
    {"name": "LLVM Discourse", "file": "llvm_discourse_posts.json", "date_field": "created_at"},
    {"name": "Apache Lists", "file": "apache_mailing_list_posts.json", "date_field": "date"},
    {"name": "Lobsters", "file": "lobsters_posts.json", "date_field": "created_at"},
    {"name": "Quora", "file": "quora_posts.json", "date_field": "answer_date"},
]


def load_posts(file_path: Path) -> list[dict]:
    """Load posts from JSON file."""
    if not file_path.exists():
        print(f"Warning: {file_path} not found. Skipping.")
        return []
    
    with open(file_path, 'r', encoding='utf-8') as f:
        data = json.load(f)
    
    # Handle both list and dict formats
    if isinstance(data, list):
        return data
    elif isinstance(data, dict) and "items" in data:
        return data["items"]
    else:
        return [data]


def normalize_date(post: dict, date_field: str) -> str | None:
    """Extract and normalize date from post."""
    date_value = post.get(date_field)
    
    if date_value is None:
        return None
    
    # Handle Unix timestamp (int/float)
    if isinstance(date_value, (int, float)):
        try:
            return datetime.fromtimestamp(date_value).strftime("%Y-%m-%d")
        except (ValueError, OSError):
            return None
    
    # Handle ISO string
    if isinstance(date_value, str):
        try:
            # Try parsing ISO format
            dt = datetime.fromisoformat(date_value.replace('Z', '+00:00'))
            return dt.strftime("%Y-%m-%d")
        except ValueError:
            # Try parsing common formats
            for fmt in ["%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%d/%m/%Y"]:
                try:
                    return datetime.strptime(date_value[:19], fmt).strftime("%Y-%m-%d")
                except ValueError:
                    continue
    
    return None


def generate_forum_distribution() -> pd.DataFrame:
    """Generate forum distribution summary."""
    results = []
    
    for forum in FORUMS:
        file_path = DATA_DIR / forum["file"]
        posts = load_posts(file_path)
        
        # Count posts with valid dates
        valid_posts = 0
        date_range_start = None
        date_range_end = None
        
        for post in posts:
            date_str = normalize_date(post, forum["date_field"])
            if date_str:
                valid_posts += 1
                if date_range_start is None or date_str < date_range_start:
                    date_range_start = date_str
                if date_range_end is None or date_str > date_range_end:
                    date_range_end = date_str
        
        results.append({
            "forum_name": forum["name"],
            "total_posts": len(posts),
            "posts_with_valid_dates": valid_posts,
            "earliest_post": date_range_start or "Unknown",
            "latest_post": date_range_end or "Unknown",
            "time_period": f"{date_range_start or 'Unknown'} to {date_range_end or 'Unknown'}",
        })
    
    return pd.DataFrame(results)


def main():
    RESULTS_DIR.mkdir(parents=True, exist_ok=True)
    
    print("Generating forum distribution...")
    df = generate_forum_distribution()
    
    # Save to CSV
    output_path = RESULTS_DIR / "forum_distribution.csv"
    df.to_csv(output_path, index=False)
    
    print(f"\nForum Distribution Summary:")
    print(df.to_string(index=False))
    print(f"\nSaved to {output_path}")
    
    return df


if __name__ == "__main__":
    main()
```

```python
"""
Phase 1 Task 2: Search Criteria Documentation
Generate search criteria summary from config file.
"""

import yaml
import pandas as pd
from pathlib import Path

PROJECT_ROOT = Path(__file__).resolve().parent.parent
CONFIG_DIR = PROJECT_ROOT / "config"
RESULTS_DIR = PROJECT_ROOT / "results" / "phase1"


def generate_search_criteria() -> pd.DataFrame:
    """Generate search criteria summary from config."""
    config_path = CONFIG_DIR / "search_criteria.yaml"
    
    with open(config_path, 'r', encoding='utf-8') as f:
        config = yaml.safe_load(f)
    
    results = []
    
    for forum_key, forum_config in config["forums"].items():
        results.append({
            "forum_name": forum_key.replace("_", " ").title(),
            "search_type": forum_config["search_type"],
            "keywords": ", ".join(forum_config.get("keywords", [])),
            "tags": ", ".join(forum_config.get("tags", [])) or "N/A",
            "filters": "; ".join(forum_config.get("filters", [])) or "N/A",
            "api_endpoint": forum_config.get("api_endpoint", "Manual"),
            "justification": forum_config.get("justification", ""),
        })
    
    return pd.DataFrame(results)


def main():
    RESULTS_DIR.mkdir(parents=True, exist_ok=True)
    
    print("Generating search criteria documentation...")
    df = generate_search_criteria()
    
    output_path = RESULTS_DIR / "search_criteria.csv"
    df.to_csv(output_path, index=False)
    
    print(f"\nSearch Criteria Summary:")
    print(df.to_string(index=False))
    print(f"\nSaved to {output_path}")
    
    return df


if __name__ == "__main__":
    main()
```

```python
"""
Phase 1 Task 3: Stack Overflow Tag Verification
Verify and extract posts containing specific JNI tags.
"""

import json
import pandas as pd
from pathlib import Path

PROJECT_ROOT = Path(__file__).resolve().parent.parent
DATA_DIR = PROJECT_ROOT / "data" / "raw"
RESULTS_DIR = PROJECT_ROOT / "results" / "phase1"

SO_TARGET_TAGS = ["java-native-interface", "jni", "jniwrapper", "jna"]


def verify_so_tags() -> pd.DataFrame:
    """Verify Stack Overflow posts contain required tags."""
    so_file = DATA_DIR / "stackoverflow_posts.json"
    
    if not so_file.exists():
        print(f"Error: {so_file} not found")
        return pd.DataFrame()
    
    with open(so_file, 'r', encoding='utf-8') as f:
        posts = json.load(f)
    
    if isinstance(posts, dict) and "items" in posts:
        posts = posts["items"]
    
    results = []
    
    for post in posts:
        post_id = post.get("question_id", "unknown")
        title = post.get("title", "")
        tags = post.get("tags", [])
        
        # Check if post has any of the target tags
        has_target_tag = any(tag in SO_TARGET_TAGS for tag in tags)
        
        # Determine which specific tags are present
        matched_tags = [tag for tag in tags if tag in SO_TARGET_TAGS]
        
        results.append({
            "post_id": post_id,
            "title": title[:100],  # Truncate for readability
            "all_tags": ", ".join(tags),
            "has_target_tag": has_target_tag,
            "matched_tags": ", ".join(matched_tags) if matched_tags else "None",
            "num_matched_tags": len(matched_tags),
        })
    
    df = pd.DataFrame(results)
    
    # Summary statistics
    print(f"\nStack Overflow Tag Verification Summary:")
    print(f"  Total posts: {len(df)}")
    print(f"  Posts with target tags: {df['has_target_tag'].sum()}")
    print(f"  Posts without target tags: {(~df['has_target_tag']).sum()}")
    
    # Count by specific tag
    print(f"\n  Tag distribution:")
    for tag in SO_TARGET_TAGS:
        count = df['matched_tags'].str.contains(tag).sum()
        print(f"    {tag}: {count} posts")
    
    return df


def main():
    RESULTS_DIR.mkdir(parents=True, exist_ok=True)
    
    print("Verifying Stack Overflow tags...")
    df = verify_so_tags()
    
    if not df.empty:
        output_path = RESULTS_DIR / "so_tag_verification.csv"
        df.to_csv(output_path, index=False)
        print(f"\nSaved to {output_path}")
    
    return df


if __name__ == "__main__":
    main()
```

```python
"""
Phase 1 Task 4: Temporal Split
Divide posts into Pre-GenAI and Post-GenAI cohorts.
"""

import json
import pandas as pd
from pathlib import Path
from datetime import datetime

PROJECT_ROOT = Path(__file__).resolve().parent.parent
DATA_DIR = PROJECT_ROOT / "data" / "raw"
RESULTS_DIR = PROJECT_ROOT / "results" / "phase1"
TEMPORAL_DIR = RESULTS_DIR / "temporal_split"

GENAI_CUTOFF_DATE = datetime(2022, 11, 30)  # ChatGPT launch date

FORUMS = [
    {"name": "Stack Overflow", "file": "stackoverflow_posts.json", "date_field": "creation_date"},
    {"name": "GitHub Issues", "file": "github_issues.json", "date_field": "created_at"},
    {"name": "Bugzilla", "file": "bugzilla_posts.json", "date_field": "creation_time"},
    {"name": "Microsoft Q&A", "file": "microsoft_qa_posts.json", "date_field": "creationDate"},
    {"name": "Oracle Community", "file": "oracle_community_posts.json", "date_field": "post_date"},
    {"name": "Reddit", "file": "reddit_posts.json", "date_field": "created_utc"},
    {"name": "Hacker News", "file": "hacker_news_posts.json", "date_field": "time"},
    {"name": "LLVM Discourse", "file": "llvm_discourse_posts.json", "date_field": "created_at"},
    {"name": "Apache Lists", "file": "apache_mailing_list_posts.json", "date_field": "date"},
    {"name": "Lobsters", "file": "lobsters_posts.json", "date_field": "created_at"},
    {"name": "Quora", "file": "quora_posts.json", "date_field": "answer_date"},
]


def normalize_date(post: dict, date_field: str) -> datetime | None:
    """Extract and normalize date from post."""
    date_value = post.get(date_field)
    
    if date_value is None:
        return None
    
    # Handle Unix timestamp
    if isinstance(date_value, (int, float)):
        try:
            return datetime.fromtimestamp(date_value)
        except (ValueError, OSError):
            return None
    
    # Handle ISO string
    if isinstance(date_value, str):
        try:
            return datetime.fromisoformat(date_value.replace('Z', '+00:00'))
        except ValueError:
            for fmt in ["%Y-%m-%dT%H:%M:%S", "%Y-%m-%d", "%d/%m/%Y"]:
                try:
                    return datetime.strptime(date_value[:19], fmt)
                except ValueError:
                    continue
    
    return None


def temporal_split() -> tuple[pd.DataFrame, pd.DataFrame]:
    """Split all posts into Pre-GenAI and Post-GenAI cohorts."""
    TEMPORAL_DIR.mkdir(parents=True, exist_ok=True)
    
    pre_genai_posts = []
    post_genai_posts = []
    
    for forum in FORUMS:
        file_path = DATA_DIR / forum["file"]
        
        if not file_path.exists():
            print(f"Warning: {file_path} not found. Skipping.")
            continue
        
        with open(file_path, 'r', encoding='utf-8') as f:
            posts = json.load(f)
        
        if isinstance(posts, dict) and "items" in posts:
            posts = posts["items"]
        
        for post in posts:
            post_date = normalize_date(post, forum["date_field"])
            
            if post_date is None:
                continue
            
            # Add metadata
            post_record = {
                "forum": forum["name"],
                "post_id": post.get("question_id") or post.get("id") or post.get("number", "unknown"),
                "title": post.get("title", "")[:100],
                "creation_date": post_date.strftime("%Y-%m-%d"),
                "era": "Pre-GenAI" if post_date < GENAI_CUTOFF_DATE else "Post-GenAI",
            }
            
            if post_date < GENAI_CUTOFF_DATE:
                pre_genai_posts.append(post_record)
            else:
                post_genai_posts.append(post_record)
    
    pre_df = pd.DataFrame(pre_genai_posts)
    post_df = pd.DataFrame(post_genai_posts)
    
    # Save to CSV
    pre_df.to_csv(TEMPORAL_DIR / "pre_genai_posts.csv", index=False)
    post_df.to_csv(TEMPORAL_DIR / "post_genai_posts.csv", index=False)
    
    print(f"\nTemporal Split Summary:")
    print(f"  Pre-GenAI posts: {len(pre_df)}")
    print(f"  Post-GenAI posts: {len(post_df)}")
    print(f"  Total posts: {len(pre_df) + len(post_df)}")
    print(f"  Cutoff date: {GENAI_CUTOFF_DATE.strftime('%Y-%m-%d')}")
    
    return pre_df, post_df


def generate_temporal_summary(pre_df: pd.DataFrame, post_df: pd.DataFrame) -> pd.DataFrame:
    """Generate summary by forum and era."""
    all_posts = pd.concat([pre_df, post_df], ignore_index=True)
    
    summary = all_posts.groupby(["forum", "era"]).size().unstack(fill_value=0)
    summary["Total"] = summary.sum(axis=1)
    summary = summary.reset_index()
    
    # Ensure both columns exist
    if "Pre-GenAI" not in summary.columns:
        summary["Pre-GenAI"] = 0
    if "Post-GenAI" not in summary.columns:
        summary["Post-GenAI"] = 0
    
    return summary[["forum", "Pre-GenAI", "Post-GenAI", "Total"]]


def main():
    RESULTS_DIR.mkdir(parents=True, exist_ok=True)
    
    print("Performing temporal split...")
    pre_df, post_df = temporal_split()
    
    print("\nGenerating temporal summary by forum...")
    summary = generate_temporal_summary(pre_df, post_df)
    
    output_path = RESULTS_DIR / "temporal_split_summary.csv"
    summary.to_csv(output_path, index=False)
    
    print(f"\nTemporal Split by Forum:")
    print(summary.to_string(index=False))
    print(f"\nSaved to {output_path}")
    
    return summary


if __name__ == "__main__":
    main()
```

```python
def generate_master_spreadsheet():
    """Generate the master spreadsheet matching professor's table structure."""
    
    # Load individual results
    forum_dist = pd.read_csv(RESULTS_DIR / "forum_distribution.csv")
    temporal_summary = pd.read_csv(RESULTS_DIR / "temporal_split_summary.csv")
    
    # Merge data
    master_data = []
    
    for _, row in forum_dist.iterrows():
        forum_name = row["forum_name"]
        
        # Get temporal data for this forum
        temporal_row = temporal_summary[temporal_summary["forum"] == forum_name]
        if not temporal_row.empty:
            pre_genai = int(temporal_row["Pre-GenAI"].values[0])
            post_genai = int(temporal_row["Post-GenAI"].values[0])
            total = int(temporal_row["Total"].values[0])
        else:
            pre_genai = 0
            post_genai = 0
            total = int(row["total_posts"])
        
        # Get search criteria
        criteria = SEARCH_CRITERIA.get(forum_name, {})
        
        master_data.append({
            "Forum Used for topic Search": forum_name,
            "Identification Technique (Rules)": criteria.get("technique", "Unknown"),
            "Pre Gen of AI": pre_genai,
            "Post Gen AI": post_genai,
            "Post Get": int(row["total_posts"]),
            "Inclusion Criteria": criteria.get("inclusion", "Unknown"),
            "Exclusion Criteria": criteria.get("exclusion", "Unknown"),
            "Total Post": total,
            "Time Period": row["time_period"],
            "Keywords": criteria.get("keywords", ""),
            "Tags": criteria.get("tags", ""),
        })
    
    df = pd.DataFrame(master_data)
    
    # Save to Excel
    output_path = RESULTS_DIR / "phase1_master_spreadsheet.xlsx"
    
    with pd.ExcelWriter(output_path, engine="openpyxl") as writer:
        df.to_excel(writer, index=False, sheet_name="Master Summary")
        
        # Auto-adjust column widths
        worksheet = writer.sheets["Master Summary"]
        for column in worksheet.columns:
            max_length = min(max(len(str(cell.value or "")) for cell in column) + 2, 50)
            worksheet.column_dimensions[column[0].column_letter].width = max_length
    
    # Also save as CSV
    csv_path = RESULTS_DIR / "phase1_master_spreadsheet.csv"
    df.to_csv(csv_path, index=False)
    
    print(f"\nMaster Spreadsheet Summary:")
    print(df.to_string(index=False))
    print(f"\nSaved to {output_path}")
    print(f"Also saved to {csv_path}")
    
    return df


def main():
    RESULTS_DIR.mkdir(parents=True, exist_ok=True)
    
    print("Generating master spreadsheet...")
    df = generate_master_spreadsheet()
    
    return df


if __name__ == "__main__":
    main()
```
